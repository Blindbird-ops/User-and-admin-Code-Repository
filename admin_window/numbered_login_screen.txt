 32:   Future<void> _checkAutoLoginOrOffline() async {
 33:     // 1. Check Connectivity
 34:     var connectivityResult = await (Connectivity().checkConnectivity());
 35:     bool isOffline = connectivityResult == ConnectivityResult.none;
 36: 
 37:     final prefs = await SharedPreferences.getInstance();
 38:     final String? cachedToken = prefs.getString('auth_token');
 39:     final String? cachedUid = prefs.getString('auth_uid');
 40:     final String? cachedEmail = prefs.getString('auth_email');
 41: 
 42:     if (isOffline) {
 43:       if (cachedUid != null && mounted) {
 44:         _showOfflineDialog(cachedToken, cachedUid);
 45:       }
 46:     } else {
 47:       // --- ONLINE AUTO-LOGIN LOGIC ---
 48:       if (cachedToken != null && cachedUid != null && cachedEmail != null) {
 49:         setState(() => isLoading = true);
 50:         var userData = await firebaseService.fetchUserData(cachedToken);
 51:         
 52:         if (userData != null && userData.isNotEmpty) {
 53:           String userRole = userData['role'] ?? 'user';
 54:           if (userRole == 'admin') {
 55:             if (mounted) {
 56:               Navigator.pushReplacement(
 57:                 context,
 58:                 MaterialPageRoute(
 59:                   builder: (context) => AdminScreen(token: cachedToken, userId: cachedUid, email: cachedEmail),
 60:                 ),
 61:               );
 62:             }
 63:           } else {
 64:             await prefs.clear();
 65:             if (mounted) setState(() => isLoading = false);
 66:           }
 67:         } else {
 68:           await prefs.clear();
 69:           if (mounted) setState(() => isLoading = false);
 70:         }
 71:       }
 72:     }

291:   void _login() async {
292:     if (!mounted) return;
293: 
294:     setState(() => isLoading = true);
295: 
296:     try {
297:       final emailInput = emailController.text.trim();
298:       final passwordInput = passwordController.text.trim();
299: 
300:       // 1. Sign in via REST API
301:       String? token = await firebaseService.signIn(emailInput, passwordInput);
302: 
303:       if (!mounted) return;
304: 
305:       if (token != null) {
306:         // 2. Sign in via Native SDK (Optional but good for Storage)
307:         try {
308:           await FirebaseAuth.instance.signInWithEmailAndPassword(
309:             email: emailInput,
310:             password: passwordInput,
311:           );
312:         } catch (e) {
313:           print("Native Auth Warning: $e");
314:         }
315: 
316:         // 3. Fetch User Data
317:         var userData = await firebaseService.fetchUserData(token);
318:         if (userData != null && userData.isNotEmpty) {
319:           String userRole = userData['role'] ?? 'user';
320:           String uid = userData['uid'] ?? '';
321:           String email = userData['email'] ?? '';
322: 
323:           if (userRole != 'admin') {
324:              ScaffoldMessenger.of(context).showSnackBar(
325:               const SnackBar(
326:                 content: Text("Access denied. You must be an admin to log in."),
327:                 backgroundColor: Colors.red,
328:               ),
329:             );
330:             setState(() => isLoading = false);
331:             return;
332:           }
333: 
334:           // --- SAVE CREDENTIALS FOR AUTO-LOGIN ---
335:           final prefs = await SharedPreferences.getInstance();
336:           await prefs.setString('auth_token', token);
337:           await prefs.setString('auth_uid', uid);
338:           await prefs.setString('auth_email', email);
339:           // ---------------------------------------
340: 
341:           await firebaseService.recordLog(token, "$email logged in", role: userRole);
342:           await firebaseService.cleanOldLogs(token);
343: 
344:           Navigator.pushReplacement(
345:             context,
346:             MaterialPageRoute(
347:               builder: (context) => AdminScreen(token: token, userId: uid, email: email,),
348:             ),
349:           );
350:         } else {
351:           setState(() => isLoading = false);
352:           ScaffoldMessenger.of(context).showSnackBar(
353:             const SnackBar(content: Text("Login Failed: Could not fetch user data.")),
354:           );
355:         }
356:       } else {
357:         setState(() => isLoading = false);
358:         ScaffoldMessenger.of(context).showSnackBar(
359:           const SnackBar(content: Text("Login Failed: Invalid token received.")),
360:         );
361:       }
362:     } catch (e) {
363:       if (!mounted) return;
364:       setState(() => isLoading = false);
365:       ScaffoldMessenger.of(context).showSnackBar(
366:         SnackBar(content: Text(e.toString())),
367:       );
368:     }
369:   }
